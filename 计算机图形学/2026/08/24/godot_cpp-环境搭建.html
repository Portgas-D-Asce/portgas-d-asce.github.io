自定义插件

my_node.h

#include <godot_cpp/classes/node.hpp>

namespace godot {
    class MyNode : public Node {
        GDCLASS(MyNode, Node)

    private:
        double speed = 0.0;

    protected:
        static void _bind_methods();

    public:
        MyNode();
        void _process(double delta) override;
        double get_speed() const;
        void set_speed(double p_speed);
    };
}

my_node.cpp

#include "my_node.h"
#include <godot_cpp/core/class_db.hpp>

using namespace godot;

void MyNode::_bind_methods() {
    ClassDB::bind_method(D_METHOD("get_speed"), &MyNode::get_speed);
    ClassDB::bind_method(D_METHOD("set_speed", "value"), &MyNode::set_speed);
    ADD_PROPERTY(PropertyInfo(Variant::FLOAT, "speed"), "set_speed", "get_speed");
    ADD_SIGNAL(MethodInfo("speed_changed", PropertyInfo(Variant::FLOAT, "new_speed")));
}

MyNode::MyNode() {}
void MyNode::_process(double delta) {
    speed += delta * 10.0;
    emit_signal("speed_changed", speed);
}
double MyNode::get_speed() const { return speed; }
void MyNode::set_speed(double p_speed) {
    speed = p_speed;
    emit_signal("speed_changed", speed);
}

// 入口函数
extern "C" {
GDExtensionBool my_extension_init(GDExtensionInterfaceGetProcAddress p_get_proc_address,
                                  GDExtensionClassLibraryPtr p_library,
                                  GDExtensionInitialization *r_initialization) {
    godot::GDExtensionBinding::InitObject init_obj(p_get_proc_address, p_library, r_initialization);
    init_obj.register_initializer([](godot::ModuleInitializationLevel p_level) {
        if (p_level == godot::MODULE_INITIALIZATION_LEVEL_SCENE) {
            ClassDB::register_class<MyNode>();
        }
    });
    return init_obj.init();
}
}

CMakeLists.txt

cmake_minimum_required(VERSION 4.3)
project(godot_cpp_demo)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 你的扩展源文件
add_library(my_extension SHARED src/my_node.cpp)

target_include_directories(my_extension PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src
        ${CMAKE_CURRENT_SOURCE_DIR}/godot_cpp/godot-cpp-4.5/gen/include
        ${CMAKE_CURRENT_SOURCE_DIR}/godot_cpp/godot-cpp-4.5/include
        ${CMAKE_CURRENT_SOURCE_DIR}/godot_cpp/godot-cpp-4.5/gdextension
)

target_link_directories(my_extension PRIVATE godot_cpp/godot-cpp-4.5/bin)

# 链接 godot-cpp 静态库
target_link_libraries(my_extension PRIVATE libgodot-cpp.linux.template_release.x86_64.a)

# 编译选项
target_compile_options(my_extension PRIVATE -Wall -Wextra)

# 设置输出名（去掉 lib 前缀，方便 Godot 加载）
set_target_properties(my_extension PROPERTIES PREFIX "" OUTPUT_NAME "my_extension")

编译即可得到产物 my_extension.so

使用插件

godot 工程，根目录下新建 bin 文件夹，将 my_extension.so 放入其中

根目录下新建 demo.gdextension 文件

[configuration]
entry_symbol = "my_extension_init"
compatibility_minimum = "4.5"

[libraries]
linux.debug.x86_64 = "res://bin/my_extension.so"
linux.release.x86_64 = "res://bin/my_extension.so"

extends Node2D

var my_node

# Called when the node enters the scene tree for the first time.
func _ready() -> void:
	my_node = MyNode.new()   # MyNode 来自 GDExtension
	add_child(my_node)
	
	# 连接信号
	my_node.connect("speed_changed", Callable(self, "_on_speed_changed"))

	# 调用方法和属性
	my_node.set_speed(30.0)
	print("当前速度：", my_node.get_speed())
	pass # Replace with function body.


# Called every frame. 'delta' is the elapsed time since the previous frame.
func _process(delta: float) -> void:
	pass
	
func _on_speed_changed(new_speed):
	print("速度变化：", new_speed)
